==================
Compile-time field reflection
==================

func inspect[T](v: T) {
    comptime for f in fields(T) {
        comptime if f.is_var {
            v.(f) = v.(f)
        } else {
            println(f.name)
        }
        comptime match f.type {
            Int -> { println("int") }
            [E] -> { println("list") }
            E? -> { println("nullable") }
            else -> comptime error("unsupported")
        }
    }
}
